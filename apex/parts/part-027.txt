function orientationScreenAxes(e){

 const gamma=Number.isFinite(e.gamma)?e.gamma:0;
 const info=screenOrientationInfo(),a=info.angle;
 let roll=gamma,pitch=beta;

 // siempre a ejes VISUALES de la pantalla: roll = izquierda/derecha; pitch = delante/atrás.

 else if(a===180){ roll=-gamma; pitch=-beta; }
 gyro.orientationType=info.type;
 adaptGyroToScreenAngle(a,roll,pitch);

}

function apexDriveSteeringConfig(speedNorm,device){
 const p=apexDriveProfile().id,d=String(device||apexInputDevice||'keyboard');
 // El modelo físico ya reduce el lock con la velocidad. Aquí sólo modulamos la entrada:
 // no volvemos a recortar el volante hasta dejar al coche sin autoridad en curvas rápidas.
 if(p==='raw')return{authority:1-.52*speedNorm,rise:mobilePerformanceMode()?lerp(11,7.4,speedNorm):lerp(5.1,2.8,speedNorm),center:mobilePerformanceMode()?13:7.2,counter:mobilePerformanceMode()?15:7.8};
 const analog=d==='gyro'||d==='gamepad';
 if(p==='simulation')return{authority:analog?lerp(1,.82,speedNorm):lerp(.99,.62,speedNorm),rise:analog?lerp(8.0,5.2,speedNorm):(d==='touch'?lerp(7.6,4.8,speedNorm):lerp(6.0,4.15,speedNorm)),center:analog?8.2:(d==='touch'?8.7:8.1),counter:analog?9.0:(d==='touch'?9.5:9.0)};
 // ASISTIDO / control clásico: tecla = intención de giro, no porcentaje de volante.
 // Mantiene bastante autoridad a alta velocidad y suaviza sólo la velocidad con que llega a ella.
 return{authority:analog?lerp(1,.88,speedNorm):lerp(1,.64,speedNorm),rise:analog?lerp(8.2,5.6,speedNorm):(d==='touch'?lerp(7.4,4.9,speedNorm):lerp(6.4,4.35,speedNorm)),center:analog?8.4:(d==='touch'?8.8:8.2),counter:analog?9.4:(d==='touch'?9.8:9.3)};
}
